# Job Applications Service Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build `apps/job-applications`, a Cloudflare Workers RPC service for tracking job applications — D1-backed records unique on `(email, applicationUrl)`, R2-backed resume PDFs, JWT-authenticated via the shared `token-auth` secret.

**Architecture:** Four-layer clean DDD with Effect as the vehicle. `domain/` is pure (aggregate + status FSM + value validation, zero framework imports); `application/` holds `Context.Tag` ports and one `Effect.gen` use case per file; `infrastructure/` provides D1/Drizzle, R2, and jose adapters as `Layer`s; `src/index.ts` is a thin `WorkerEntrypoint` that Schema-decodes input, verifies the JWT, provides a per-request Layer, runs the use case, and maps errors to an `RpcResult` envelope (errors cross the RPC boundary as data).

**Tech Stack:** TypeScript 6 (strictest preset), Effect 3.22 (`Effect`/`Layer`/`Context`/`Schema`/`DateTime`), Drizzle ORM 0.45 on Cloudflare D1, `uuid` v7, `jose` 6 (HS256), Vitest 4 + `@cloudflare/vitest-pool-workers` (real D1/R2 in tests), pnpm workspaces, Node >= 24, ESM.

**Spec:** `docs/superpowers/specs/2026-08-16-job-applications-service-design.md` — the plan argues from the spec; executors read both.

## Global Constraints

- Node `>=24`, `"type": "module"`; imports use explicit `.ts` extensions (repo convention).
- Extend `packages/tsconfig/tsconfig.base.json`; exclude `coverage`; `types: ["node"]`; `ignoreDeprecations: "6.0"`.
- ESLint: `@ethang/eslint-config` (`config.main.js`) — rules include `no-try-catch` (use Effect error channel, `Effect.tryPromise`), `no-barrel-files` (no `export *` re-export files), `prefer-lodash`. The lint script runs `eslint . --fix` then `tsc --noEmit`; autofix may strip explicit return type annotations — do not re-add them.
- TDD (AGENTS.md rule 2): every step pair is failing-test → implementation; red → green → refactor. Tests are finite-state machines: `it.each` over the full input domain (valid, invalid, boundary, terminal states).
- Coverage: 100% target; CI gate 80%. `vitest run --coverage` with v8 provider, `include: ["src/**/*.ts"]`, `exclude: ["src/**/*.d.ts"]`.
- The owner `email` ALWAYS comes from the verified JWT (`token-auth` HS256 secret) — never from a request parameter. Unique constraint `(email, applicationUrl)`.
- RPC methods return `RpcResult<T>`: `{ ok: true; value: T } | { ok: false; error: { code; message } }`. Error codes: `VALIDATION`, `UNAUTHENTICATED`, `NOT_FOUND`, `DUPLICATE`, `INVALID_TRANSITION`, `RESUME`, `INTERNAL`.
- Resume: key `${email}/${id}`, ≤ 5 MB, `%PDF` magic prefix required; re-upload overwrites in place.
- Status FSM: `applied → screening → interview → offer` via `cycleStatus`; `offer`/`rejected`/`withdrawn` are terminal (`INVALID_TRANSITION`). `updateApplication` may set any valid status.
- Commit after every task with a conventional message (`feat:`, `test:`, `chore:`).

```ts
export const STATUSES = ["applied", "screening", "interview", "offer", "rejected", "withdrawn"] as const;

export type Status = (typeof STATUSES)[number];

const NEXT: Readonly<Record<Status, null | Status>> = {
  applied: "screening",
  screening: "interview",
  interview: "offer",
  offer: null,
  rejected: null,
  withdrawn: null
};

export const isStatus = (value: string): value is Status => {
  return (STATUSES as readonly string[]).includes(value);
};

export const nextStatus = (status: Status): null | Status => {
  return NEXT[status];
};
```

```ts
import { Effect } from "effect";

import { InvalidStatusTransitionError } from "../../errors/invalid-status-transition-error.ts";
import { ValidationError } from "../../errors/validation-error.ts";

  applicationUrl: "https://example.com/jobs/1",
  appliedDate: "2026-08-01",
  company: "Acme",
  email: "me@example.com",
  title: "Engineer"
};

const run = <A>(effect: Effect.Effect<A, unknown>) => Effect.runSync(effect);

const make = (overrides: Partial<JobApplication> = {}) => {
  return run(createJobApplication(VALID_INPUT)) as JobApplication & Partial<JobApplication>;
};

describe("createJobApplication", () => {
  it("creates with defaults and timestamps", () => {
    const app = run(createJobApplication(VALID_INPUT));
    expect(app.id).toMatch(/^[0-9a-f]{8}-/);
    expect(app.status).toBe("applied");
    expect(app.location).toBeNull();
    expect(app.resumeKey).toBeNull();
    expect(app.createdAt).toBe(app.updatedAt);
    expect(app.email).toBe("me@example.com");
  });

  it.each(["company", "title", "applicationUrl"])("rejects empty %s", (field) => {
    const result = Effect.runSync(Effect.flip(createJobApplication({ ...VALID_INPUT, [field]: "" })));
    expect(result).toBeInstanceOf(ValidationError);
  });

  it.each(["2026-8-1", "01/08/2026", "not-a-date", "2026-13-01"])("rejects malformed appliedDate %s", (appliedDate) => {

  });

    );
    expect(result).toBeInstanceOf(ValidationError);
  });

  it("honors an explicit status and optional fields", () => {
    const app = run(
      createJobApplication({ ...VALID_INPUT, status: "interview", location: "Remote", salary: "$150k", notes: "referral" })

    expect(app.status).toBe("interview");
    expect(app.location).toBe("Remote");
    expect(app.salary).toBe("$150k");
    expect(app.notes).toBe("referral");

    ["applied", "screening"],
    ["screening", "interview"],
    ["interview", "offer"]
```

```ts
const nowIso = () => {
  return DateTime.formatIso(DateTime.unsafeNow());
};

const isIsoDate = (value: string) => {
  return /^\d{4}-\d{2}-\d{2}$/.test(value);
};

const requireText = (value: string, field: string): Effect.Effect<string, ValidationError> => {
  if (isEmpty(value)) {
    return Effect.fail(new ValidationError(`${field} must not be empty`));
  }
  return Effect.succeed(value);
};

const requireStatus = (value: Status): Effect.Effect<Status, ValidationError> => {
  if (!isStatus(value)) {
    return Effect.fail(new ValidationError(`invalid status: ${value}`));
  }
  return Effect.succeed(value);
};

const requireDate = (value: string): Effect.Effect<string, ValidationError> => {
  if (!isIsoDate(value)) {
    return Effect.fail(new ValidationError("appliedDate must be YYYY-MM-DD"));
  }
  return Effect.succeed(value);
};

const optional = <T>(value: null | T | undefined): null | T => {
  return isNil(value) ? null : value;
};

export const createJobApplication = (input: CreateApplicationInput) => {
  return Effect.gen(function* () {
    const company = yield* requireText(input.company, "company");
    const title = yield* requireText(input.title, "title");
    const applicationUrl = yield* requireText(input.applicationUrl, "applicationUrl");
    const email = yield* requireText(input.email, "email");
    const appliedDate = yield* requireDate(input.appliedDate);
    const status = isNil(input.status) ? "applied" : yield* requireStatus(input.status);
    const now = nowIso();
    const application: JobApplication = {
      applicationUrl,
      appliedDate,
      company,
      createdAt: now,
      email,
      id: v7(),
      location: optional(input.location),
      nextInterviewDate: optional(input.nextInterviewDate),
      notes: optional(input.notes),
      resumeFilename: null,
      resumeKey: null,
      resumeSize: null,
      salary: optional(input.salary),
      status,
      title,
      updatedAt: now
    };
    return application;
  });
};

export const withChanges = (application: JobApplication, changes: UpdateApplicationChanges) => {
  return Effect.gen(function* () {
    const hasChanges =
      !isNil(changes.appliedDate) ||
      !isNil(changes.company) ||
      !isNil(changes.location) ||
      !isNil(changes.nextInterviewDate) ||
      !isNil(changes.notes) ||
      !isNil(changes.salary) ||
      !isNil(changes.status) ||
      !isNil(changes.title);
    if (!hasChanges) {
      return yield* Effect.fail(new ValidationError("no changes provided"));
    }
    const company = isNil(changes.company) ? application.company : yield* requireText(changes.company, "company");
    const title = isNil(changes.title) ? application.title : yield* requireText(changes.title, "title");
    const appliedDate = isNil(changes.appliedDate) ? application.appliedDate : yield* requireDate(changes.appliedDate);
    const status = isNil(changes.status) ? application.status : yield* requireStatus(changes.status);
    return {
      ...application,
      appliedDate,
      company,
      location: optional(changes.location),
      nextInterviewDate: optional(changes.nextInterviewDate),
      notes: optional(changes.notes),
      salary: optional(changes.salary),
      status,
      title,
      updatedAt: nowIso()
    };
  });
};

export const advanceStatus = (application: JobApplication) => {
  const next = nextStatus(application.status);
  if (next === null) {
    return Effect.fail(new InvalidStatusTransitionError(`cannot advance from ${application.status}`));
  }
  return Effect.succeed({ ...application, status: next, updatedAt: nowIso() });
};
```

```ts
import { Context, Effect, Layer } from "effect";

import type { JobApplication } from "../../domain/job-application/aggregate.ts";
import { DuplicateApplicationError } from "../../errors/duplicate-application-error.ts";
import { JobApplicationRepository } from "../ports.ts";

type Repository = Context.Tag.Service<typeof JobApplicationRepository>;

export const createFakeRepository = (initial: readonly JobApplication[] = []) => {
  const rows = new Map<string, JobApplication>();
  for (const row of initial) {
    rows.set(row.id, row);
  }
  const repository: Repository = {
    delete: (id, email) => {
      const row = rows.get(id);
      if (row === undefined || row.email !== email) {
        return Effect.succeed(false);
      }
      rows.delete(id);
      return Effect.succeed(true);
    },
    findByEmailAndUrl: (email, applicationUrl) => {
      for (const row of rows.values()) {
        if (row.email === email && row.applicationUrl === applicationUrl) {
          return Effect.succeed(row);
        }
      }
      return Effect.succeed(null);
    },
    findById: (id, email) => {
      const row = rows.get(id);
      if (row === undefined || row.email !== email) {
        return Effect.succeed(null);
      }
      return Effect.succeed(row);
    },
    insert: (application) => {
      for (const row of rows.values()) {
        if (row.email === application.email && row.applicationUrl === application.applicationUrl) {
          return Effect.fail(new DuplicateApplicationError("application already exists"));
        }
      }
      rows.set(application.id, application);
      return Effect.succeed(application);
    },
    list: ({ after, email, first, status }) => {
      let items = [...rows.values()].filter(
        (row) => row.email === email && (status === null || row.status === status)
      );
      if (after !== null) {
        items = items.filter((row) => row.id < after);
      }
      items = items.sort((a, b) => b.id.localeCompare(a.id)).slice(0, first);
      return Effect.succeed(items);
    },
    update: (application) => {
      rows.set(application.id, application);
      return Effect.succeed(application);
    }
  };
  return { layer: Layer.succeed(JobApplicationRepository, repository), rows };
};
```

```ts
import { Effect } from "effect";

import { attachResume } from "../domain/job-application/aggregate.ts";
import { NotFoundError } from "../errors/not-found-error.ts";
import { ResumeError } from "../errors/resume-error.ts";
import { JobApplicationRepository, ResumeStore } from "./ports.ts";

const MAX_RESUME_BYTES = 5 * 1024 * 1024;
const PDF_MAGIC = "%PDF";

export const uploadResume = (parameters: {
  readonly data: ArrayBuffer;
  readonly email: string;
  readonly filename: string;
  readonly id: string;
}) => {
  return Effect.gen(function* () {
    const repo = yield* JobApplicationRepository;
    const store = yield* ResumeStore;
    const existing = yield* repo.findById(parameters.id, parameters.email);
    if (existing === null) {
      return yield* Effect.fail(new NotFoundError("application not found"));
    }
    const bytes = new Uint8Array(parameters.data);
    if (bytes.byteLength > MAX_RESUME_BYTES) {
      return yield* Effect.fail(new ResumeError("resume exceeds 5 MB"));
    }
    const header = new TextDecoder().decode(bytes.subarray(0, PDF_MAGIC.length));
    if (header !== PDF_MAGIC) {
      return yield* Effect.fail(new ResumeError("resume must be a PDF"));
    }
    const key = `${parameters.email}/${parameters.id}`;
    yield* store.put(key, parameters.data, parameters.filename);
    const updated = attachResume(existing, { filename: parameters.filename, key, size: bytes.byteLength });
    return yield* repo.update(updated);
  });
};
```

```ts
import { Effect } from "effect";

import { DuplicateApplicationError } from "./errors/duplicate-application-error.ts";
import { FetchError } from "./errors/fetch-error.ts";
import { InvalidStatusTransitionError } from "./errors/invalid-status-transition-error.ts";
import { NotFoundError } from "./errors/not-found-error.ts";
import { ResumeError } from "./errors/resume-error.ts";
import { SaveError } from "./errors/save-error.ts";
import { TokenError } from "./errors/token-error.ts";
import { ValidationError } from "./errors/validation-error.ts";

export type ErrorCode =
  | "DUPLICATE"
  | "INTERNAL"
  | "INVALID_TRANSITION"
  | "NOT_FOUND"
  | "RESUME"
  | "UNAUTHENTICATED"
  | "VALIDATION";

export type RpcResult<T> =
  | { readonly ok: true; readonly value: T }
  | { readonly ok: false; readonly error: { readonly code: ErrorCode; readonly message: string } };

const failure = (code: ErrorCode, error: { readonly message: string }): RpcResult<never> => {
  return { error: { code, message: error.message }, ok: false };
};

export const toResult = <T>(effect: Effect.Effect<T, unknown>) => {
  return Effect.match(effect, {
    onFailure: (error) => {
      if (error instanceof ValidationError) {
        return failure("VALIDATION", error);
      }
      if (error instanceof TokenError) {
        return failure("UNAUTHENTICATED", error);
      }
      if (error instanceof NotFoundError) {
        return failure("NOT_FOUND", error);
      }
      if (error instanceof DuplicateApplicationError) {
        return failure("DUPLICATE", error);
      }
      if (error instanceof InvalidStatusTransitionError) {
        return failure("INVALID_TRANSITION", error);
      }
      if (error instanceof ResumeError) {
        return failure("RESUME", error);
      }
      if (error instanceof FetchError || error instanceof SaveError) {
        return failure("INTERNAL", error);
      }
      return failure("INTERNAL", new Error(error instanceof Error ? error.message : String(error)));
    },
    onSuccess: (value) => {
      return { ok: true as const, value };
    }
  });
};
```

```ts
import { and, desc, eq, lt } from "drizzle-orm";
import { drizzle } from "drizzle-orm/d1";
import { Effect, Layer } from "effect";

import { JobApplicationRepository } from "../../application/ports.ts";
import { jobApplicationsTable } from "../../db/schema.ts";
import type { JobApplication } from "../../domain/job-application/aggregate.ts";
import { isStatus, type Status } from "../../domain/job-application/status.ts";
import { DuplicateApplicationError } from "../../errors/duplicate-application-error.ts";
import { FetchError } from "../../errors/fetch-error.ts";
import { SaveError } from "../../errors/save-error.ts";

type Row = typeof jobApplicationsTable.$inferSelect;

const parseStatus = (value: string): Status => {
  if (isStatus(value)) {
    return value;
  }
  throw new Error(`Unknown status: ${value}`);
};

const toAggregate = (row: Row): JobApplication => {
  return {
    applicationUrl: row.applicationUrl,
    appliedDate: row.appliedDate,
    company: row.company,
    createdAt: row.createdAt,
    email: row.email,
    id: row.id,
    location: row.location,
    nextInterviewDate: row.nextInterviewDate,
    notes: row.notes,
    resumeFilename: row.resumeFilename,
    resumeKey: row.resumeKey,
    resumeSize: row.resumeSize,
    salary: row.salary,
    status: parseStatus(row.status),
    title: row.title,
    updatedAt: row.updatedAt
  };
};

const toRow = (application: JobApplication): Row => {
  return {
    applicationUrl: application.applicationUrl,
    appliedDate: application.appliedDate,
    company: application.company,
    createdAt: application.createdAt,
    email: application.email,
    id: application.id,
    location: application.location,
    nextInterviewDate: application.nextInterviewDate,
    notes: application.notes,
    resumeFilename: application.resumeFilename,
    resumeKey: application.resumeKey,
    resumeSize: application.resumeSize,
    salary: application.salary,
    status: application.status,
    title: application.title,
    updatedAt: application.updatedAt
  };
};

const isUniqueViolation = (cause: unknown) => {
  const message = cause instanceof Error ? cause.message : String(cause);
  return message.includes("UNIQUE constraint failed");
};

export const createJobApplicationRepositoryLayer = (database: D1Database) => {
  const db = drizzle(database, { schema: { jobApplicationsTable } });
  return Layer.succeed(JobApplicationRepository, {
    delete: (id, email) => {
      return Effect.tryPromise({
        catch: (cause) => {
          return new FetchError(String(cause));
        },
        try: async () => {
          const result = await db
            .delete(jobApplicationsTable)
            .where(and(eq(jobApplicationsTable.id, id), eq(jobApplicationsTable.email, email)))
            .run();
          return result.meta.changes > 0;
        }
      });
    },
    findByEmailAndUrl: (email, applicationUrl) => {
      return Effect.tryPromise({
        catch: (cause) => {
          return new FetchError(String(cause));
        },
        try: async () => {
          const row = await db.query.jobApplicationsTable.findFirst({
            where: (table, operators) => {
              return operators.and(operators.eq(table.email, email), operators.eq(table.applicationUrl, applicationUrl));
            }
          });
          return row ? toAggregate(row) : null;
        }
      });
    },
    findById: (id, email) => {
      return Effect.tryPromise({
        catch: (cause) => {
          return new FetchError(String(cause));
        },
        try: async () => {
          const row = await db.query.jobApplicationsTable.findFirst({
            where: (table, operators) => {
              return operators.and(operators.eq(table.id, id), operators.eq(table.email, email));
            }
          });
          return row ? toAggregate(row) : null;
        }
      });
    },
    insert: (application) => {
      return Effect.tryPromise({
        catch: (cause) => {
          if (isUniqueViolation(cause)) {
            return new DuplicateApplicationError("application already exists");
          }
          return new SaveError(String(cause));
        },
        try: async () => {
          const [row] = await db.insert(jobApplicationsTable).values(toRow(application)).returning();
          if (!row) {
            throw new Error("insert returned no rows");
          }
          return toAggregate(row);
        }
      });
    },
    list: ({ after, email, first, status }) => {
      return Effect.tryPromise({
        catch: (cause) => {
          return new FetchError(String(cause));
        },
        try: async () => {
          const conditions = [eq(jobApplicationsTable.email, email)];
          if (status !== null) {
            conditions.push(eq(jobApplicationsTable.status, status));
          }
          if (after !== null) {
            conditions.push(lt(jobApplicationsTable.id, after));
          }
          const rows = await db
            .select()
            .from(jobApplicationsTable)
            .where(and(...conditions))
            .orderBy(desc(jobApplicationsTable.id))
            .limit(first);
          return rows.map(toAggregate);
        }
      });
    },
    update: (application) => {
      return Effect.tryPromise({
        catch: (cause) => {
          return new SaveError(String(cause));
        },
        try: async () => {
          await db.update(jobApplicationsTable).set(toRow(application)).where(eq(jobApplicationsTable.id, application.id)).run();
          return application;
        }
      });
    }
  });
};
```

```ts
import { Effect, Layer } from "effect";
import { jwtVerify } from "jose";
import isNil from "lodash/isNil.js";

import { TokenVerifier } from "../../application/ports.ts";
import { TokenError } from "../../errors/token-error.ts";

export const createTokenVerifierLayer = (secret: string) => {
  const secretKey = new TextEncoder().encode(secret);
  return Layer.succeed(TokenVerifier, {
    verify: (token) => {
      return Effect.tryPromise({
        catch: (cause) => {
          return new TokenError(String(cause));
        },
        try: async () => {
          const { payload } = await jwtVerify(token, secretKey);
          const email = payload.email;
          if (isNil(email) || typeof email !== "string") {
            throw new Error("token is missing email claim");
          }
          return email;
        }
      });
    }
  });
};
```

```ts
import { WorkerEntrypoint } from "cloudflare:workers";
import { Effect, Layer, Schema } from "effect";

import { createApplication } from "./application/create-application.ts";
import { cycleStatus } from "./application/cycle-status.ts";
import { deleteApplication } from "./application/delete-application.ts";
import { getApplication } from "./application/get-application.ts";
import { getResume } from "./application/get-resume.ts";
import { listApplications } from "./application/list-applications.ts";
import { TokenVerifier } from "./application/ports.ts";
import {
  CreateApplicationInputSchema,
  decodeInput,
  ListApplicationsParamsSchema,
  UpdateApplicationChangesSchema
} from "./application/schemas.ts";
import { updateApplication } from "./application/update-application.ts";
import { uploadResume } from "./application/upload-resume.ts";
import { createJobApplicationRepositoryLayer } from "./infrastructure/drizzle/repository.ts";
import { createResumeStoreLayer } from "./infrastructure/r2/resume-store.ts";
import { createTokenVerifierLayer } from "./infrastructure/token/verifier.ts";
import { ValidationError } from "./errors/validation-error.ts";
import { toResult, type RpcResult } from "./rpc-result.ts";

export class JobApplicationsService extends WorkerEntrypoint<Env> {
  private readonly layer = () => {
    return Layer.mergeAll(
      createJobApplicationRepositoryLayer(this.env.jobApplications),
      createResumeStoreLayer(this.env.jobResumes),
      createTokenVerifierLayer(this.env["token-auth"])
    );
  };

  private readonly run = <A>(program: Effect.Effect<A, unknown>): Promise<RpcResult<A>> => {
    return Effect.runPromise(toResult(program).pipe(Effect.provide(this.layer())));
  };

  private readonly verify = (token: string) => {
    return Effect.gen(function* () {
      const verifier = yield* TokenVerifier;
      return yield* verifier.verify(token);
    });
  };

  public async createApplication(parameters: unknown) {
    const verify = this.verify;
    const run = this.run;
    return run(
      Effect.gen(function* () {
        const input = yield* decodeInput(CreateApplicationInputSchema, parameters);
        const { token, ...rest } = input;
        const email = yield* verify(token);
        return yield* createApplication({ ...rest, email });
      })
    );
  }

  public async getApplication(parameters: unknown) {
    const verify = this.verify;
    const run = this.run;
    return run(
      Effect.gen(function* () {
        const input = yield* decodeInput(
          Schema.Struct({ id: Schema.NonEmptyString, token: Schema.NonEmptyString }),
          parameters
        );
        const email = yield* verify(input.token);
        return yield* getApplication(input.id, email);
      })
    );
  }

  public async listApplications(parameters: unknown) {
    const verify = this.verify;
    const run = this.run;
    return run(
      Effect.gen(function* () {
        const input = yield* decodeInput(ListApplicationsParamsSchema, parameters);
        const email = yield* verify(input.token);
        const first = Math.min(100, Math.max(1, input.first));
        return yield* listApplications({ after: input.after ?? null, email, first, status: input.status ?? null });
      })
    );
  }

  public async updateApplication(parameters: unknown) {
    const verify = this.verify;
    const run = this.run;
    return run(
      Effect.gen(function* () {
        const input = yield* decodeInput(UpdateApplicationChangesSchema, parameters);
        const { id, token, ...changes } = input;
        const email = yield* verify(token);
        return yield* updateApplication(id, email, changes);
      })
    );
  }

  public async cycleStatus(parameters: unknown) {
    const verify = this.verify;
    const run = this.run;
    return run(
      Effect.gen(function* () {
        const input = yield* decodeInput(
          Schema.Struct({ id: Schema.NonEmptyString, token: Schema.NonEmptyString }),
          parameters
        );
        const email = yield* verify(input.token);
        return yield* cycleStatus(input.id, email);
      })
    );
  }

  public async deleteApplication(parameters: unknown) {
    const verify = this.verify;
    const run = this.run;
    return run(
      Effect.gen(function* () {
        const input = yield* decodeInput(
          Schema.Struct({ id: Schema.NonEmptyString, token: Schema.NonEmptyString }),
          parameters
        );
        const email = yield* verify(input.token);
        return yield* deleteApplication(input.id, email);
      })
    );
  }

  public async uploadResume(parameters: unknown) {
    const verify = this.verify;
    const run = this.run;
    return run(
      Effect.gen(function* () {
        if (
          typeof parameters !== "object" ||
          parameters === null ||
          typeof (parameters as { data?: unknown }).data !== "object" ||
          typeof (parameters as { filename?: unknown }).filename !== "string" ||
          typeof (parameters as { id?: unknown }).id !== "string" ||
          typeof (parameters as { token?: unknown }).token !== "string"
        ) {
          return yield* Effect.fail(new ValidationError("invalid uploadResume parameters"));
        }
        const { data, filename, id, token } = parameters as { data: ArrayBuffer; filename: string; id: string; token: string };
        const email = yield* verify(token);
        return yield* uploadResume({ data, email, filename, id });
      })
    );
  }

  public async getResume(parameters: unknown) {
    const verify = this.verify;
    const run = this.run;
    return run(
      Effect.gen(function* () {
        const input = yield* decodeInput(
          Schema.Struct({ id: Schema.NonEmptyString, token: Schema.NonEmptyString }),
          parameters
        );
        const email = yield* verify(input.token);
        return yield* getResume(input.id, email);
      })
    );
  }

  public override fetch(_request: Request) {
    return new Response("OK", { status: 200 });
  }
}

export default JobApplicationsService;
```
